-- ROLE TABLE
CREATE TABLE role (
    Role_ID INT NOT NULL AUTO_INCREMENT,
    Role_Name VARCHAR(50) NOT NULL,
    PRIMARY KEY (Role_ID)
);

-- SPECIES TABLE
CREATE TABLE species (
    Species_ID INT NOT NULL AUTO_INCREMENT,
    Species_Name VARCHAR(50) NOT NULL,
    PRIMARY KEY (Species_ID)
);

-- STAFF TABLE
CREATE TABLE staff (
    Staff_ID INT NOT NULL AUTO_INCREMENT,
    F_Name VARCHAR(50) NOT NULL,
    L_Name VARCHAR(50) NOT NULL,
    Role_ID INT,
    PRIMARY KEY (Staff_ID),
    FOREIGN KEY (Role_ID) REFERENCES role(Role_ID)
);

-- ADOPTER TABLE 
CREATE TABLE adopter (
    Adopter_ID INT NOT NULL AUTO_INCREMENT,
    F_Name VARCHAR(50) NOT NULL,
    L_Name VARCHAR(50) NOT NULL,
    Email VARCHAR(100) NOT NULL,
    Address VARCHAR(255),
    PRIMARY KEY (Adopter_ID),
    UNIQUE (Email)
);

-- ADOPTER_PHONE TABLE 
CREATE TABLE adopter_phone (
    Phone_Number VARCHAR(15) NOT NULL,
    Adopter_ID INT NOT NULL,
    PRIMARY KEY (Phone_Number, Adopter_ID),
    FOREIGN KEY (Adopter_ID) REFERENCES adopter(Adopter_ID)
        ON DELETE CASCADE
);

-- BREED TABLE (moved before animal to avoid FK error)
CREATE TABLE breed (
    Breed_ID INT NOT NULL AUTO_INCREMENT,
    Breed_Name VARCHAR(50) NOT NULL,
    Species_ID INT,
    PRIMARY KEY (Breed_ID),
    FOREIGN KEY (Species_ID) REFERENCES species(Species_ID)
);

-- ANIMAL TABLE 
CREATE TABLE animal (
    Animal_ID INT NOT NULL AUTO_INCREMENT,
    Name VARCHAR(50) NOT NULL,
    Gender VARCHAR(10),
    DateOfBirth DATE,
    Breed_ID INT,
    PRIMARY KEY (Animal_ID),
    FOREIGN KEY (Breed_ID) REFERENCES breed(Breed_ID)
);

-- MEDICAL_RECORD TABLE
CREATE TABLE medical_record (
    Record_ID INT NOT NULL AUTO_INCREMENT,
    Treatment VARCHAR(255) NOT NULL,
    Treatment_Date DATE,
    Notes TEXT,
    Animal_ID INT,
    Staff_ID INT,
    PRIMARY KEY (Record_ID),
    FOREIGN KEY (Animal_ID) REFERENCES animal(Animal_ID),
    FOREIGN KEY (Staff_ID) REFERENCES staff(Staff_ID)
);

-- ADOPTION TABLE
CREATE TABLE adoption (
    Adoption_ID INT NOT NULL AUTO_INCREMENT,
    Adoption_Date DATE NOT NULL,
    Fee DECIMAL(10,2),
    Adopter_ID INT,
    Animal_ID INT,
    Staff_ID INT,
    PRIMARY KEY (Adoption_ID),
    FOREIGN KEY (Adopter_ID) REFERENCES adopter(Adopter_ID),
    FOREIGN KEY (Animal_ID) REFERENCES animal(Animal_ID),
    FOREIGN KEY (Staff_ID) REFERENCES staff(Staff_ID)
);

-- ADOPTIONRETURN TABLE
CREATE TABLE adoptionreturn (
    Return_ID INT NOT NULL AUTO_INCREMENT,
    Adoption_ID INT,
    Return_Date DATE,
    Return_Reason TEXT,
    PRIMARY KEY (Return_ID),
    UNIQUE (Adoption_ID),
    FOREIGN KEY (Adoption_ID) REFERENCES adoption(Adoption_ID)
        ON DELETE CASCADE
);

-- PAYMENT TABLE 
CREATE TABLE payment (
    Payment_ID INT NOT NULL AUTO_INCREMENT,
    Amount DECIMAL(10,2) NOT NULL,
    Payment_Date DATE NOT NULL,
    Adoption_ID INT,
    PRIMARY KEY (Payment_ID),
    FOREIGN KEY (Adoption_ID) REFERENCES adoption(Adoption_ID)
);
